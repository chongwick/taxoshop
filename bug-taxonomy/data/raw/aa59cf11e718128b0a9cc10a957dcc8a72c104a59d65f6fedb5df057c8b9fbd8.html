#include "sqlite3.h"

#include <errno.h>
#include <inttypes.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include <sys/mman.h>
#include <unistd.h>

#define SQLITE_STRLEN30_MASK UINT64_C(0x3fffffff)
#define ONE_GIB UINT64_C(0x40000000)
#define PREFIX_LEN 5u

static int make_large_uri(char **uri_out, size_t *len_out){
  size_t len = (size_t)(ONE_GIB + PREFIX_LEN);
  char *uri = mmap(NULL, len + 1, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
  if( uri==MAP_FAILED ){
    fprintf(stderr, "mmap(%zu) failed: errno=%d\n", len + 1, errno);
    return 1;
  }

  memset(uri, 'A', len);
  memcpy(uri, "file:", PREFIX_LEN);
  uri[len] = '\0';

  *uri_out = uri;
  *len_out = len;
  return 0;
}

int main(void){
  char *uri = NULL;
  size_t len = 0;
  sqlite3 *db = NULL;

  if( make_large_uri(&uri, &len)!=0 ){
    return 2;
  }

  fprintf(stderr,
          "prepared URI strlen=%zu sqlite3Strlen30-equivalent=%" PRIu64 "\n",
          len, ((uint64_t)len & SQLITE_STRLEN30_MASK));
  fflush(stderr);

  int rc = sqlite3_open_v2(uri, &db,
                           SQLITE_OPEN_READWRITE
                               | SQLITE_OPEN_CREATE
                               | SQLITE_OPEN_URI,
                           NULL);
  fprintf(stderr, "sqlite3_open_v2 returned rc=%d db=%p\n", rc, (void *)db);

  if( db!=NULL ){
    sqlite3_close(db);
  }
  munmap(uri, len + 1);
  return rc==SQLITE_OK ? 0 : 1;
}
