    def __init__(self):
        # Use ASCII digits so float() takes the fast path that expects a NUL terminator.
        self.data = [ord('1'), ord('2')]
    def __iter__(self):
        return self
    def __next__(self):
        if self.data:
            return self.data.pop(0)
        raise StopIteration
    def __length_hint__(self):
        return 0
