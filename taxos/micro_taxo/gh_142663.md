## Reproducer

```python
from array import array
import struct

a = array("d", [1.0, 2.0])
b = array("l", [1, 2])
mv = memoryview(a)
mv2 = memoryview(b)

class EvilStruct(struct.Struct):
    def unpack_from(self, buf, /, offset=0):
        mv.release()
        a.append(3.14)  # resizes array (reallocates & frees old data)
        return (1,)

struct.Struct = EvilStruct
print(mv == mv2)
```
