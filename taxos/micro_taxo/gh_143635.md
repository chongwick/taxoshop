## Reproducer

import types


class Zap:
    __slots__ = ("container",)

    def __init__(self, container):
        self.container = container

    def __getattr__(self, name):
        if name == "__origin__":
            self.container.clear()
            return None
        if name == "__args__":
            return ()
        raise AttributeError


params = []
params.append(Zap(params))
alias = types.GenericAlias(list, (params,))
