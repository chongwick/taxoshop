## Reproducer

```python
import mmap, os, tempfile

fd, path = tempfile.mkstemp()
os.write(fd, b"A" * 4096)
mm = mmap.mmap(fd, 4096, access=mmap.ACCESS_WRITE)
mv = memoryview(mm)

class Trigger:
    def __index__(self):
        mv.release()
        os.ftruncate(fd, 0)
        return 0

try:
    sub = mv[slice(Trigger(), None, None)]
    sub[0] # Trigger the slicing to call __index__
finally:
    mm.close()
    os.close(fd)
    os.unlink(path)
```
