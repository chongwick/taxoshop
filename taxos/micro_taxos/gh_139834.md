    echo "--- Configuring and building in $BUILD_DIR ---"
    mkdir -p "$BUILD_DIR"
    cd "$BUILD_DIR"
    
    CC=clang CXX=clang++ ../../configure --with-pydebug \
        --with-address-sanitizer \
